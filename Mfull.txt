--- M3.6 RSYSLOG server (HQ-SRV) + klienty ---

# SHAG 1 - HQ-SRV
apt-get install -y rsyslog
systemctl enable --now rsyslog

# SHAG 2 - proverit tekushij konfig
grep -n "imudp\|imtcp" /etc/rsyslog.conf

# SHAG 3 - raskомmentirovat UDP i TCP
sed -i 's/^#module(load="imudp")/module(load="imudp")/' /etc/rsyslog.conf
sed -i 's/^#input(type="imudp" port="514")/input(type="imudp" port="514")/' /etc/rsyslog.conf
sed -i 's/^#module(load="imtcp")/module(load="imtcp")/' /etc/rsyslog.conf
sed -i 's/^#input(type="imtcp" port="514")/input(type="imtcp" port="514")/' /etc/rsyslog.conf

# SHAG 4 - proverit chto raskommentirovalos
grep -n "imudp\|imtcp" /etc/rsyslog.conf

# SHAG 5 - shablon dlya sohraneniya logov
mkdir -p /opt
cat > /etc/rsyslog.d/remote.conf << 'EOF'
$template RemoteLogs,"/opt/%HOSTNAME%/%PROGRAMNAME%.log"
*.warn ?RemoteLogs
& stop
EOF

# SHAG 6 - pereзapустit i proverit port
systemctl restart rsyslog
ss -lnup | grep 514   # dolzhen byt 0.0.0.0:514

# esli port ne slushaet - smotret oshibki
journalctl -u rsyslog -n 20

# Rotaciya logov
cat > /etc/logrotate.d/remote-logs << 'EOF'
/opt/*/*.log {
    weekly
    compress
    missingok
    notifempty
    minsize 10M
    sharedscripts
    postrotate
        systemctl reload rsyslog
    endscript
}
EOF

# KLIENTY: HQ-RTR, BR-RTR, BR-SRV - odinakovo na vsех
apt-get install -y rsyslog
echo '*.warn @@IP_HQ-SRV:514' > /etc/rsyslog.d/client.conf
systemctl restart rsyslog

# Test
logger -p warning "test from $(hostname)"
sleep 2

# Na HQ-SRV proverit
ls /opt/
# dolzhny poyavitsya papki: hq-rtr/ br-rtr/ br-srv/
